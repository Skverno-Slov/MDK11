CREATE VIEW film_info_view AS
SELECT 
    f.film_id AS код,
    f.title AS название,
    f.release_year AS год_выхода,
    CONCAT_WS(' ', 
              FLOOR(f.duration/60), 'ч', 
              f.duration % 60, 'м'
    ) AS длительность,
    STRING_AGG(g.genre_name, ', ') AS жанр,
    f.description AS описание,
    f.rental_start_date AS начало_проката
FROM films f
LEFT JOIN film_genres fg ON f.film_id = fg.film_id
LEFT JOIN genres g ON fg.genre_id = g.genre_id
GROUP BY 
    f.film_id, 
    f.title, 
    f.release_year, 
    f.duration, 
    f.description, 
    f.rental_start_date;

CREATE VIEW upcoming_films_view AS
SELECT 
    f.film_id AS код,
    f.title AS название,
    f.release_year AS год_выхода,
    CONCAT(
        FLOOR(f.duration/60), ' ч ', 
        f.duration % 60, ' м'
    ) AS длительность,
    STRING_AGG(g.genre_name, ', ') WITHIN GROUP (ORDER BY g.genre_name) AS жанр,
    f.description AS описание,
    f.rental_start_date AS начало_проката
FROM films f
LEFT JOIN film_genres fg ON f.film_id = fg.film_id
LEFT JOIN genres g ON fg.genre_id = g.genre_id
WHERE f.rental_start_date BETWEEN GETDATE() AND DATEADD(MONTH, 1, GETDATE())
GROUP BY 
    f.film_id, 
    f.title, 
    f.release_year, 
    f.duration, 
    f.description, 
    f.rental_start_date
ORDER BY f.rental_start_date;
